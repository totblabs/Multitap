cmake_minimum_required(VERSION 3.20)
project(MultitapPixelConversionQa LANGUAGES CXX)
if(NOT PIXEL_CONVERSION_BASELINE OR NOT EXISTS "${PIXEL_CONVERSION_BASELINE}/baseline.hpp")
  message(FATAL_ERROR "Provide the frozen inspected Host conversion baseline directory")
endif()
add_executable(pixel-conversion-qa conversion.cpp)
target_compile_features(pixel-conversion-qa PRIVATE cxx_std_20)
target_include_directories(pixel-conversion-qa PRIVATE "${PIXEL_CONVERSION_BASELINE}")
target_compile_definitions(pixel-conversion-qa PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(pixel-conversion-qa PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /O2)
