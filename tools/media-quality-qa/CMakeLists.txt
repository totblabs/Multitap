cmake_minimum_required(VERSION 3.20)
project(MultitapMediaQualityQA LANGUAGES CXX)
set(HOST_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/windows_host/src")
set(CLIENT_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/windows_client/windows/runner")
add_executable(media-quality-qa quality.cpp "${HOST_SOURCE}/media.cpp" "${CLIENT_SOURCE}/remote_decoder.cpp")
target_include_directories(media-quality-qa PRIVATE "${HOST_SOURCE}" "${CLIENT_SOURCE}")
target_compile_features(media-quality-qa PRIVATE cxx_std_20)
target_compile_definitions(media-quality-qa PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(media-quality-qa PRIVATE /W4 /WX /permissive- /EHsc /utf-8 /O2)
target_link_libraries(media-quality-qa PRIVATE mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 gdi32 user32 uuid d3d11 dxgi windowsapp)
