cmake_minimum_required(VERSION 3.20)
project(MultitapMediaBench LANGUAGES CXX)
if(NOT MEDIA_SOURCE)
  set(MEDIA_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../apps/windows_host/src")
endif()
add_executable(media-benchmark benchmark.cpp "${MEDIA_SOURCE}/media.cpp")
target_include_directories(media-benchmark PRIVATE "${MEDIA_SOURCE}")
target_compile_features(media-benchmark PRIVATE cxx_std_20)
target_compile_definitions(media-benchmark PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
target_compile_options(media-benchmark PRIVATE /W4 /permissive- /EHsc /utf-8 /O2)
target_link_libraries(media-benchmark PRIVATE mfplat mf mfuuid wmcodecdspuuid ole32 oleaut32 gdi32 user32 uuid d3d11 dxgi windowsapp)
