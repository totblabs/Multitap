cmake_minimum_required(VERSION 3.20)
project(ThumbnailPrivacyIO LANGUAGES CXX)
set(runner "${CMAKE_CURRENT_SOURCE_DIR}/../../../windows/runner")
add_executable(thumbnail_privacy_io fixture.cpp "${runner}/thumbnail_writer.cpp")
target_include_directories(thumbnail_privacy_io PRIVATE "${runner}")
target_compile_features(thumbnail_privacy_io PRIVATE cxx_std_17)
target_compile_definitions(thumbnail_privacy_io PRIVATE NOMINMAX)
target_compile_options(thumbnail_privacy_io PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
target_link_libraries(thumbnail_privacy_io PRIVATE windowscodecs ole32 oleaut32 shell32 user32 advapi32)
