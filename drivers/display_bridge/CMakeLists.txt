cmake_minimum_required(VERSION 3.24)
project(MultitapVirtualDisplayBridge LANGUAGES CXX)
if(NOT IS_DIRECTORY "${VDD_SDK_SOURCE}")
  message(FATAL_ERROR "Provide the inspected, pinned libvirtualdisplay source directory")
endif()
set(BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SUNSHINE_VIRTUAL_DISPLAY_DRIVER OFF CACHE BOOL "" FORCE)
set(BUILD_VIRTUALDISPLAY_TOOLS OFF CACHE BOOL "" FORCE)
set(BUILD_VIRTUALDISPLAY_PROBE OFF CACHE BOOL "" FORCE)
set(BUILD_VIRTUALDISPLAY_VULKAN_LAYER OFF CACHE BOOL "" FORCE)
add_subdirectory("${VDD_SDK_SOURCE}" sdk EXCLUDE_FROM_ALL)
add_executable(MultitapVirtualDisplayBridgeTest provider_test.cpp)
add_executable(MultitapNativeDisplayRuntimeTest runtime_test.cpp)
add_executable(MultitapNativeDisplayLeaseRecoveryTest lease_recovery_test.cpp)
target_compile_features(MultitapNativeDisplayLeaseRecoveryTest PRIVATE cxx_std_20)
target_link_libraries(MultitapNativeDisplayLeaseRecoveryTest PRIVATE libvirtualdisplay::driver)
target_compile_features(MultitapNativeDisplayRuntimeTest PRIVATE cxx_std_20)
target_link_libraries(MultitapNativeDisplayRuntimeTest PRIVATE libvirtualdisplay::driver)
target_compile_features(MultitapVirtualDisplayBridgeTest PRIVATE cxx_std_20)
target_link_libraries(MultitapVirtualDisplayBridgeTest PRIVATE libvirtualdisplay::driver)
if(MSVC)
  set_property(TARGET MultitapNativeDisplayLeaseRecoveryTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(MultitapNativeDisplayLeaseRecoveryTest PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  set_property(TARGET MultitapVirtualDisplayBridgeTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(MultitapVirtualDisplayBridgeTest PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
  set_property(TARGET MultitapNativeDisplayRuntimeTest PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_compile_options(MultitapNativeDisplayRuntimeTest PRIVATE /W4 /WX /permissive- /EHsc /utf-8)
endif()
